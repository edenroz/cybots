<script>

document.addEventListener('DOMContentLoaded', () => {


    /* ======================================================
       MATRIX RAIN ENGINE
       ====================================================== */

    const canvas = document.createElement('canvas');

    canvas.id = 'cy-matrix-bg';

    canvas.style.cssText =
        'position:fixed;' +
        'top:0;' +
        'left:0;' +
        'width:100vw;' +
        'height:100vh;' +
        'z-index:0;' +
        'pointer-events:none;' +
        'opacity:0.42;';

    document.body.prepend(canvas);


    const ctx = canvas.getContext('2d');


    let width =
        canvas.width =
        window.innerWidth;

    let height =
        canvas.height =
        window.innerHeight;


    /* Resize */

    window.addEventListener('resize', () => {

        width =
            canvas.width =
            window.innerWidth;

        height =
            canvas.height =
            window.innerHeight;

    });


    /* Matrix characters */

    const chars =
        '01CY_BORG_ERR0R_SYSTEM_FAILURE_OVERRIDE_10101011_CYBERPUNK_SYS_KILL_';


    const fontSize = 14;


    let columns =
        Math.floor(width / fontSize);


    let drops =
        Array(columns).fill(1);


    /* Recalculate columns on resize */

    window.addEventListener('resize', () => {

        columns =
            Math.floor(
                canvas.width / fontSize
            );

        drops =
            Array(columns).fill(1);

    });


    /* Draw Matrix */

    function drawMatrix() {

        ctx.fillStyle =
            'rgba(0, 0, 0, 0.08)';

        ctx.fillRect(
            0,
            0,
            width,
            height
        );


        ctx.font =
            fontSize +
            'px "Space Mono", monospace';


        for (
            let i = 0;
            i < drops.length;
            i++
        ) {

            const isMagenta =
                Math.random() < 0.08;


            ctx.fillStyle =
                isMagenta
                    ? '#ff0055'
                    : '#0000ff';


            const text =
                chars.charAt(
                    Math.floor(
                        Math.random() *
                        chars.length
                    )
                );


            ctx.fillText(
                text,
                i * fontSize,
                drops[i] * fontSize
            );


            if (
                drops[i] * fontSize > height &&
                Math.random() > 0.975
            ) {

                drops[i] = 0;

            }


            drops[i]++;

        }

    }


    setInterval(
        drawMatrix,
        73
    );


    /* ======================================================
       TEXT → INDIVIDUAL LETTERS
       ====================================================== */

    function glitchifyTextNode(textNode) {

        const text =
            textNode.nodeValue;


        if (
            !text ||
            !text.trim()
        ) {
            return;
        }


        const fragment =
            document.createDocumentFragment();


        for (
            const char of text
        ) {


            /*
             * Preserve whitespace.
             */

            if (
                /\s/.test(char)
            ) {

                fragment.appendChild(
                    document.createTextNode(char)
                );

                continue;

            }


            const span =
                document.createElement('span');


            span.className =
                'cy-glitch-letter';


            span.textContent =
                char;


            /*
             * Random delay.
             */

            span.style.setProperty(
                '--glitch-delay',
                (
                    Math.random() * 5
                ).toFixed(2) + 's'
            );


            fragment.appendChild(
                span
            );

        }


        textNode.parentNode.replaceChild(
            fragment,
            textNode
        );

    }


    /* ======================================================
       PROCESS ELEMENT
       ====================================================== */

    function processElement(element) {


        /*
         * Already processed?
         */

        if (
            element.classList.contains(
                'cy-glitch-processed'
            )
        ) {

            return;

        }


        /*
         * Never touch user cards.
         */

        if (
            element.closest(
                '.UserCard-profile'
            )
        ) {

            return;

        }


        /*
         * Find text nodes only.
         *
         * This means links, images,
         * strong, em, etc. remain intact.
         */

        const walker =
            document.createTreeWalker(
                element,
                NodeFilter.SHOW_TEXT,
                {

                    acceptNode(node) {

                        if (
                            !node.nodeValue.trim()
                        ) {

                            return NodeFilter.FILTER_REJECT;

                        }


                        const parent =
                            node.parentElement;


                        if (!parent) {

                            return NodeFilter.FILTER_REJECT;

                        }


                        /*
                         * Never process:
                         *
                         * - existing glitch letters
                         * - script
                         * - style
                         * - code
                         * - pre
                         */

                        if (
                            parent.closest(
                                '.cy-glitch-letter,' +
                                'script,' +
                                'style,' +
                                'code,' +
                                'pre'
                            )
                        ) {

                            return NodeFilter.FILTER_REJECT;

                        }


                        return NodeFilter.FILTER_ACCEPT;

                    }

                }
            );


        const nodes = [];


        let node;


        while (
            node = walker.nextNode()
        ) {

            nodes.push(node);

        }


        /*
         * Convert every text node
         * into individual letters.
         */

        nodes.forEach(
            glitchifyTextNode
        );


        element.classList.add(
            'cy-glitch-processed'
        );

    }


    /* ======================================================
       APPLY GLITCH
       ====================================================== */

    function applySafeGlitch() {


        /*
         * TITLES
         */

        const titles =
            document.querySelectorAll(
                '.Hero h2,' +
                '.DiscussionListItem-title,' +
                '.PostUser-name a,' +
                '.Header-title a'
            );


        titles.forEach(
            element => {

                if (
                    element.closest(
                        '.UserCard-profile'
                    )
                ) {

                    return;

                }


                processElement(
                    element
                );

            }
        );


        /*
         * POSTS
         */

        const posts =
            document.querySelectorAll(
                '.Post-body p'
            );


        posts.forEach(
            processElement
        );

    }


    /* ======================================================
       RANDOM LETTER GLITCH
       ====================================================== */

    function triggerRandomGlitch() {

        const letters = Array.from(
            document.querySelectorAll(
                '.cy-glitch-letter'
            )
        );


        if (!letters.length) {
            return;
        }


        /*
         * Numero di gruppi.
         *
         * 20% → 2 gruppi
         * 80% → 1 gruppo
         */

        const groupCount =
            Math.random() < 0.20 ? 2 : 1;


        for (
            let g = 0;
            g < groupCount;
            g++
        ) {

            /*
             * Punto di partenza casuale.
             */

            const start =
                Math.floor(
                    Math.random() *
                    letters.length
                );


            /*
             * Dimensione del gruppo.
             *
             * 65% → 1-3
             * 25% → 4-7
             * 10% → 8-12
             */

            const roll =
                Math.random();


            let size;


            if (
                roll < 0.65
            ) {

                size =
                    1 +
                    Math.floor(
                        Math.random() * 3
                    );

            } else if (
                roll < 0.90
            ) {

                size =
                    4 +
                    Math.floor(
                        Math.random() * 4
                    );

            } else {

                size =
                    8 +
                    Math.floor(
                        Math.random() * 5
                    );

            }


            const selected =
                letters.slice(
                    start,
                    start + size
                );


            /*
             * Corrupt group.
             */

            selected.forEach(
                (letter, index) => {

                    /*
                     * Reset eventuale stato precedente.
                     */

                    letter.classList.remove(
                        'cy-glitch-active',
                        'cy-glitch-magenta',
                        'cy-glitch-blue',
                        'cy-glitch-rgb'
                    );


                    /*
                     * Force reflow.
                     */

                    void letter.offsetWidth;


                    /*
                     * Tipo di corruzione.
                     */

                    const mode =
                        Math.random();


                    if (
                        mode < 0.20
                    ) {

                        /*
                         * Magenta
                         */

                        letter.classList.add(
                            'cy-glitch-magenta'
                        );

                    } else if (
                        mode < 0.40
                    ) {

                        /*
                         * Blu
                         */

                        letter.classList.add(
                            'cy-glitch-blue'
                        );

                    } else if (
                        mode < 0.65
                    ) {

                        /*
                         * RGB split
                         */

                        letter.classList.add(
                            'cy-glitch-rgb'
                        );

                    }


                    /*
                     * Carattere corrotto.
                     */

                    if (
                        Math.random() < 0.38
                    ) {

                        letter.dataset.original =
                            letter.textContent;


                        const corruptedChars = [
                            '█',
                            '▓',
                            '▒',
                            '░',
                            'Ø',
                            'X',
                            '#',
                            '?',
                            '0',
                            '1',
                            'Ξ',
                            '╳',
                            '╱',
                            '╲',
                            '¤',
                            '§'
                        ];


                        letter.textContent =
                            corruptedChars[
                                Math.floor(
                                    Math.random() *
                                    corruptedChars.length
                                )
                            ];

                    }


                    /*
                     * Piccolo ritardo progressivo
                     * all'interno del gruppo.
                     */

                    letter.style.setProperty(
                        '--glitch-delay',
                        (
                            index * 0.012
                        ) + 's'
                    );


                    letter.classList.add(
                        'cy-glitch-active'
                    );


                    /*
                     * Ripristino.
                     */

                    setTimeout(
                        () => {

                            if (
                                letter.dataset.original
                            ) {

                                letter.textContent =
                                    letter.dataset.original;

                                delete letter.dataset.original;

                            }


                            letter.classList.remove(
                                'cy-glitch-active',
                                'cy-glitch-magenta',
                                'cy-glitch-blue',
                                'cy-glitch-rgb'
                            );

                        },
                        180 + index * 15
                    );

                }
            );

        }

    }


    /* ======================================================
       RANDOM GLITCH SCHEDULER
       
       Glitch normali frequenti +
       occasionali burst da 2-3 glitch.
       ====================================================== */

    function scheduleRandomGlitch() {

        /*
         * 25% di probabilità di creare un burst.
         */

        const isBurst =
            Math.random() < 0.25;


        if (
            isBurst
        ) {

            /*
             * 2 oppure 3 glitch ravvicinati.
             */

            const burstCount =
                2 +
                Math.floor(
                    Math.random() * 2
                );


            let burstIndex = 0;


            function runBurst() {

                triggerRandomGlitch();

                burstIndex++;


                if (
                    burstIndex < burstCount
                ) {

                    /*
                     * 70–180 ms
                     * tra i glitch del burst.
                     */

                    const burstDelay =
                        70 +
                        Math.random() * 110;


                    setTimeout(
                        runBurst,
                        burstDelay
                    );

                } else {

                    /*
                     * Dopo il burst:
                     * pausa più lunga.
                     *
                     * 800–2500 ms
                     */

                    const nextDelay =
                        800 +
                        Math.random() * 1700;


                    setTimeout(
                        scheduleRandomGlitch,
                        nextDelay
                    );

                }

            }


            runBurst();

        } else {

            /*
             * Glitch normale.
             *
             * 400–1800 ms
             */

            triggerRandomGlitch();


            const nextDelay =
                400 +
                Math.random() * 1400;


            setTimeout(
                scheduleRandomGlitch,
                nextDelay
            );

        }

    }


    /*
     * Start random glitch scheduler.
     */

    scheduleRandomGlitch();


    /* ======================================================
       INITIAL APPLICATION
       ====================================================== */

    applySafeGlitch();


    /* ======================================================
       FLARUM DYNAMIC CONTENT
       ====================================================== */

    let processing =
        false;


    const observer =
        new MutationObserver(
            () => {


                if (
                    processing
                ) {

                    return;

                }


                processing =
                    true;


                requestAnimationFrame(
                    () => {

                        applySafeGlitch();

                        processing =
                            false;

                    }
                );

            }
        );


    observer.observe(
        document.body,
        {
            childList: true,
            subtree: true
        }
    );

});

</script>
