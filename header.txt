<script>
document.addEventListener('DOMContentLoaded', () => {
  // --- MATRIX RAIN BACKGROUND ---
  const canvas = document.createElement('canvas');
  canvas.id = 'cy-matrix-bg';
  canvas.style.cssText = 'position:fixed;top:0;left:0;width:100vw;height:100vh;z-index:0;pointer-events:none;opacity:0.42;';
  document.body.prepend(canvas);

  const ctx = canvas.getContext('2d');
  let width = canvas.width = window.innerWidth;
  let height = canvas.height = window.innerHeight;

  window.addEventListener('resize', () => {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
  });

  const chars = '01CY_BORG_ERR0R_SYSTEM_FAILURE_OVERRIDE_10101011_CYBERPUNK_SYS_KILL_';
  const fontSize = 14;
  let columns = Math.floor(width / fontSize);
  let drops = Array(columns).fill(1);

  // Ricalcola le colonne quando ruoti il telefono
  window.addEventListener('resize', () => {
    columns = Math.floor(canvas.width / fontSize);
    drops = Array(columns).fill(1);
  });

  function drawMatrix() {
    ctx.fillStyle = 'rgba(0, 0, 0, 0.08)';
    ctx.fillRect(0, 0, width, height);
    ctx.font = fontSize + 'px "Space Mono", monospace';

    for (let i = 0; i < drops.length; i++) {
      const isMagenta = Math.random() < 0.08;
      ctx.fillStyle = isMagenta ? '#ff0055' : '#0000ff';

      const text = chars.charAt(Math.floor(Math.random() * chars.length));
      ctx.fillText(text, i * fontSize, drops[i] * fontSize);

      if (drops[i] * fontSize > height && Math.random() > 0.975) {
        drops[i] = 0;
      }
      drops[i]++;
    }
  }
  setInterval(drawMatrix, 73);

  // --- GLITCH APPLICATION ENGINE (Safe Mode) ---
  const applySafeGlitch = () => {
    // Applica ai titoli
    const titles = document.querySelectorAll('.Hero h2, .DiscussionListItem-title, .PostUser-name a, .Header-title a');
    titles.forEach(el => {
      if (el.closest('.UserCard-profile')) return;
      if (!el.classList.contains('cy-safe-glitch')) {
        el.classList.add('cy-safe-glitch');
      }
    });

    // Applica ai post (senza rompere i link o le immagini all'interno!)
    const posts = document.querySelectorAll('.Post-body p');
    posts.forEach(p => {
      if (p.closest('.UserCard-profile')) return;
      if (!p.classList.contains('cy-safe-glitch-post')) {
        p.classList.add('cy-safe-glitch-post');
      }
    });
  };

  // Esegui all'avvio
  applySafeGlitch();

  // Osserva dinamicamente senza far impazzire Flarum
  const observer = new MutationObserver(() => {
    applySafeGlitch();
  });
  observer.observe(document.body, { childList: true, subtree: true });
});
</script>
